cmake_minimum_required(VERSION 3.16)
project(mlprep_cpp CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MLPREP_SOLUTIONS "Build tests against the reference solutions" OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra -O2)
set(SRC_DIR ${CMAKE_CURRENT_SOURCE_DIR}/exercises)
if(MLPREP_SOLUTIONS)
  set(SRC_DIR ${CMAKE_CURRENT_SOURCE_DIR}/solutions)
endif()
enable_testing()
file(GLOB TESTS ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_*.cpp)
foreach(t ${TESTS})
  get_filename_component(name ${t} NAME_WE)
  add_executable(${name} ${t})
  target_include_directories(${name} PRIVATE ${SRC_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/tests)
  find_package(Threads)
  target_link_libraries(${name} PRIVATE Threads::Threads)
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES TIMEOUT 30)
endforeach()
